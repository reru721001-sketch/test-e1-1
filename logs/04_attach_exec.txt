## 2. exec: 새 프로세스로 들어갔다 나오기
$ docker run -di --name ubuntu-ae ubuntu:24.04 bash
bb4722f4e99c74c7907a0af3e8f3f422b697e16a2fed1900588dc15c1bbc68c5

$ printf "echo inside-via-exec\nps -e -o pid,cmd\nexit\n" | docker exec -i ubuntu-ae bash
inside-via-exec
    PID CMD
      1 bash
      7 bash
     13 ps -e -o pid,cmd

$ docker ps --filter name=ubuntu-ae
CONTAINER ID   IMAGE          COMMAND   CREATED        STATUS                  PORTS     NAMES
bb4722f4e99c   ubuntu:24.04   "bash"    1 second ago   Up Less than a second             ubuntu-ae

## 3. attach: 메인 프로세스(PID 1)에 붙었다 나오기
$ printf "echo inside-via-attach; echo PID=\$\$\nexit\n" | docker attach ubuntu-ae

$ docker logs ubuntu-ae
inside-via-attach
PID=1

$ docker ps -a --filter name=ubuntu-ae
CONTAINER ID   IMAGE          COMMAND   CREATED         STATUS                              PORTS     NAMES
bb4722f4e99c   ubuntu:24.04   "bash"    2 seconds ago   Exited (0) Less than a second ago             ubuntu-ae
